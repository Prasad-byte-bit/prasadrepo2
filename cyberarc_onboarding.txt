---
- name: Get API Token and Onboard Accounts to CyberArk
  hosts: myservers
  gather_facts: false

  vars_files:
    - vars/system_config.yml

  vars:
    script_failed: false
    script_error_reason: ""

  tasks:

    - name: Read user credentials from CSV
      community.general.read_csv:
        path: "{{ csv_path }}"
      register: user_data
      failed_when: user_data.list is not defined or user_data.list | length == 0
      ignore_errors: true

    - name: Check if CSV read failed
      set_fact:
        script_failed: true
        script_error_reason: "CSV file is missing or empty"
      when: user_data.failed is defined and user_data.failed

    - name: Show user data from CSV
      debug:
        var: user_data.list
      when: not script_failed

    - name: Get Auth Token from CyberArk
      uri:
        url: "{{ api_url }}"
        method: POST
        headers:
          Content-Type: "application/json"
        body_format: json
        body:
          username: "{{ api_user }}"
          password: "{{ api_pass }}"
        return_content: true
        status_code: 200
        validate_certs: false
      register: token_response
      failed_when: token_response.status != 200 or ('content' not in token_response)
      ignore_errors: true
      when: not script_failed

    - name: Check if token fetch failed
      set_fact:
        script_failed: true
        script_error_reason: "Failed to fetch authentication token"
      when: token_response.failed is defined and token_response.failed

    - name: Parse token JSON
      set_fact:
        auth_token: "{{ (token_response.content | from_json).token }}"
      when: not script_failed and token_response.content is defined
      failed_when: auth_token is not defined
      ignore_errors: true

    - name: Check if token parse failed
      set_fact:
        script_failed: true
        script_error_reason: "Token parsing failed"
      when: auth_token is not defined or auth_token | length == 0

    - name: Show parsed token
      debug:
        msg: "Token received: {{ auth_token }}"
      when: not script_failed

    - name: Onboard accounts to CyberArk
      block:
        - name: Onboard account
          uri:
            url: "{{ cyberark_api_url }}"
            method: POST
            headers:
              Content-Type: "application/json"
              Authorization: "{{ auth_token }}"
            body_format: json
            body:
              name: "{{ item.username }}"
              address: "{{ account_address }}"
              userName: "{{ item.username }}"
              platformId: "{{ platform_id }}"
              safeName: "{{ safe_name }}"
              secret: "{{ item.password }}"
            return_content: true
            status_code: [200, 201]
            validate_certs: false
          loop: "{{ user_data.list }}"
          loop_control:
            label: "{{ item.username }}"
          register: onboard_results
          failed_when: onboard_results.failed
          ignore_errors: true
      when: not script_failed
      rescue:
        - set_fact:
            script_failed: true
            script_error_reason: "Account onboarding API failed"

    - name: Show onboarding responses
      debug:
        var: onboard_results.results
      when: onboard_results is defined and not script_failed

    # Final Task — Show success or failure message
    - name: Print final script status
      debug:
        msg: >-
          {{
            ' Script executed successfully.' if not script_failed
            else ' Script failed: ' ~ script_error_reason
          }}
